# Re-run of the 8B stages of the freeze check, and a stop-and-resume test of the pod disk
# layout (work on the container disk, one tarball per finished stage). Plan: the handover of
# 2026-11-26. Runbook: TEST_SESSION_RUNBOOK.md, section "Resume after a failure".
#
# What it proves. Run 1 stops on purpose at resume-probe, after the chain packed the build
# stage. The pod stays up. Run 2 resumes: the chain skips vram8-build because its tarball
# matches, and check-resume proves it. The 8B commands equal freeze_rerun_stages.txt.
#
# Two launches on one pod, the same --pod-id and --stages:
#   1. Fresh, with --keep-pod-on-fail (KEEP_POD_ON_FAIL=1). resume-probe exits 1 the first
#      time, the chain stops and the watcher reports the pod as kept.
#   2. With --resume (RESUME=1). The chain runs setup, context and gpulog again, skips
#      vram8-build, then runs the rest. The pod stops when every stage is ok.
#
# Tarball tags (the chain counts the stage lines from 1): vram8-build is stage 4, so its
# tarball is 04-vram8-build.tgz. A test pins the tag, so a reorder fails the test.
# No hosted call. About 45 min and about $0.70 for both launches.

SWEEP_GGUF_DIR=/root/gguf

!setup@60m bash prefreeze/pod_setup35.sh --commit "$COMMIT" --inputs $WORK/inputs --passages-dir $WORK/passages --ragas-venv $WORK/ragas-venv --ref-venv /root/exp35-ref-venv
!context@5m uv run --no-sync python context35.py > "$OUT_DIR/context35.json"
gpulog@1m nohup nvidia-smi --query-gpu=timestamp,memory.used,memory.total,utilization.gpu --format=csv -lms 1000 > "$OUT_DIR/gpu.csv" 2>&1 & echo $! >> "$BACKGROUND_PIDS"

vram8-build@20m uv run --no-sync python run_arms.py --dataset freshstack --index e8b-c1024 --stage ingest --limit 5 --corpus-sample 150 --passages-dir $WORK/passages --out-dir "$OUT_DIR/vram" --prefreeze --local-dir /root/exp35-local && uv run --no-sync python run_arms.py --dataset freshstack --index e8b-c1024 --stage seal --limit 5 --corpus-sample 150 --passages-dir $WORK/passages --out-dir "$OUT_DIR/vram" --prefreeze --local-dir /root/exp35-local
!resume-probe@1m test -e "$WORK/resume-probe.ok" || { touch "$WORK/resume-probe.ok"; echo "resume-probe: stopping the chain on purpose (first run)"; exit 1; }
vram8-search@20m uv run --no-sync python run_arms.py --dataset freshstack --setup e8b-c1024-r8b-hon --stage search --limit 5 --corpus-sample 150 --passages-dir $WORK/passages --index-from "$OUT_DIR/vram" --out-dir "$OUT_DIR/vramq" --prefreeze --local-dir /root/exp35-local

check-search@5m uv run --no-sync python prefreeze/verify35.py expect "$OUT_DIR/vramq/freshstack/search/e8b-c1024-r8b-hon/run1/search-summary.json" --count 1 'closed_missing=[]'
check-tokenizer@5m uv run --no-sync python prefreeze/verify35.py expect "$OUT_DIR/vram/freshstack/e8b-c1024/index-fingerprint.json" --count 1 tokenizer.sha256=83cdf8c3a34f68862319cb1810ee7b1e2c0a44e0864ae930194ddb76bb7feb8d tokenizer.revision=5cf2132abc99cad020ac570b19d031efec650f2b
check-resume@2m grep -q '== vram8-build: SKIPPED (resume' "$WORK/test-session.log" && cd "$WORK/stage-out" && sha256sum -c 04-vram8-build.tgz.sha256
collect@5m mkdir -p "$OUT_DIR/router-logs" && cd /root/exp35-local && find . -name server.log -print0 | tar --null -T - -cf - | tar -xmf - -C "$OUT_DIR/router-logs"
